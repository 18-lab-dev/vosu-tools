cmake_minimum_required(VERSION 3.22)
project(VosuRealm LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(REALM_VERSION 20.1.2)
set(REALM_ENABLE_SYNC OFF CACHE BOOL "" FORCE)
set(REALM_ENABLE_ENCRYPTION OFF CACHE BOOL "" FORCE)
set(REALM_BUILD_LIB_ONLY ON CACHE BOOL "" FORCE)
set(REALM_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(REALM_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(REALM_BUILD_OBJECT_STORE OFF CACHE BOOL "" FORCE)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/source")
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/realm-core-20.1.2.tar.gz" ARCHIVE_HASH)
set(OLD_HASH "")
if(EXISTS "${CMAKE_CURRENT_BINARY_DIR}/source.hash")
 file(READ "${CMAKE_CURRENT_BINARY_DIR}/source.hash" OLD_HASH)
endif()
if(NOT OLD_HASH STREQUAL ARCHIVE_HASH)
 file(ARCHIVE_EXTRACT INPUT "${CMAKE_CURRENT_SOURCE_DIR}/realm-core-20.1.2.tar.gz" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/source")
 file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/source.hash" "${ARCHIVE_HASH}")
endif()
add_subdirectory("${CMAKE_CURRENT_BINARY_DIR}/source" core EXCLUDE_FROM_ALL)
add_library(vosu_realm STATIC bridge.cpp)
if(WIN32)
 target_compile_definitions(vosu_realm PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
endif()
target_link_libraries(vosu_realm PRIVATE Storage)
install(TARGETS vosu_realm Storage ARCHIVE DESTINATION lib)
